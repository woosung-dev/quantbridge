//@version=5
strategy("MA Crossover Strategy", overlay=true, initial_capital=10000,
         default_qty_type=strategy.percent_of_equity, default_qty_value=90,
         commission_type=strategy.commission.percent, commission_value=0.055)

// === 파라미터 ===
fastLength  = input.int(12, "빠른 이동평균", minval=1, maxval=100)
slowLength  = input.int(30, "느린 이동평균", minval=2, maxval=400)
atrLength   = input.int(14, "ATR 길이", minval=1)
atrMult     = input.float(2.0, "ATR 손절 배수", minval=0.5, step=0.1)
useShort    = input.bool(true, "숏 진입 허용")

// === 지표 ===
fastMa = ta.sma(close, fastLength)
slowMa = ta.sma(close, slowLength)
atr    = ta.atr(atrLength)

// === 시그널 ===
crossUp   = ta.crossover(fastMa, slowMa)
crossDown = ta.crossunder(fastMa, slowMa)

// === 손절 가격 ===
var float longStop  = na
var float shortStop = na

if crossUp
    longStop := close - atr * atrMult
    strategy.entry("Long", strategy.long)

if crossDown and useShort
    shortStop := close + atr * atrMult
    strategy.entry("Short", strategy.short)

// === 청산 ===
if strategy.position_size > 0
    strategy.exit("Long Exit", from_entry="Long", stop=longStop)
    if crossDown
        strategy.close("Long", comment="시그널 청산")

if strategy.position_size < 0
    strategy.exit("Short Exit", from_entry="Short", stop=shortStop)
    if crossUp
        strategy.close("Short", comment="시그널 청산")

// === 플롯 ===
plot(fastMa, "Fast MA", color=color.orange, linewidth=2)
plot(slowMa, "Slow MA", color=color.blue, linewidth=2)
plotshape(crossUp, style=shape.triangleup, location=location.belowbar)
